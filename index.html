#include int main() { int students, i; float marks[100], average; int belowAverage = 0; float sum = 0; printf("===== Student Performance Risk Alert System =====\n"); printf("Enter number of students: "); scanf("%d", &students); for(i = 0; i < students; i++) { printf("Enter marks of student %d: ", i + 1); scanf("%f", &marks[i]); sum += marks[i]; } average = sum / students; printf("\nAverage Marks = %.2f\n", average); printf("\nStudents Below Average:\n"); for(i = 0; i < students; i++) { if(marks[i] < average) { printf("Student %d is at Risk (Marks = %.2f)\n", i + 1, marks[i]); belowAverage++; } } printf("\nTotal Students at Risk: %d\n", belowAverage); return 0; }
